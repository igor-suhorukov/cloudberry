option('pg_config', type: 'string', value: 'pg_config',
       description: 'pg_config of the patched PostgreSQL 19 to build against')

option('orca', type: 'boolean', value: true,
       description: 'Build gp_orca (the ORCA optimizer module)')
option('orca_debug', type: 'feature', value: 'auto',
       description: 'ORCA\'s own debug checks, GPOS_DEBUG; auto: where the server checks its assertions, as Cloudberry\'s --enable-cassert')

option('pax', type: 'boolean', value: true,
       description: 'Build pax, Cloudberry\'s PAX table access method (C++; needs protobuf and liburing)')

option('udp2', type: 'boolean', value: false,
       description: 'Build the udp2 interconnect transport (decision 6: after it stabilises)')

option('tde', type: 'boolean', value: false,
       description: 'Build gp_tde (decision 8: only on a formal requirement for in-database encryption)')

option('require_patched_pg', type: 'boolean', value: true,
       description: 'Fail configuration if the core patch series is missing from PostgreSQL')

option('hook_tests', type: 'boolean', value: true,
       description: 'Build gp_probe, the module the hook tests use to exercise the core patch series')
option('gpfdist', type: 'feature', value: 'auto',
       description: 'Build gpfdist, the file server of external tables; it needs APR and libevent, and takes transforms where libyaml is installed')
option('gpfts', type: 'feature', value: 'auto',
       description: 'Build gpfts, the coordinator\'s automatic failover; it needs libcurl and jansson, and etcd where it runs')
option('gpmgmt', type: 'boolean', value: true,
       description: 'Install gpMgmt, Cloudberry\'s management tools, beside the server (Python 3 with PyGreSQL and psutil at run time)')
